Browser dibatasi oleh sandbox keamanan sehingga tidak bisa melakukan koneksi TCP Telnet secara langsung. Langkah koneksi nyata:

  1. Instal Node.js di komputer lokal Anda.
  2. Buat file bernama olt-proxy.js dan salin kode Node.js berikut:
const express = require('express');
const { Telnet } = require('telnet-client');
const cors = require('cors');
const app = express();
app.use(cors()); app.use(express.json());
app.post('/api/olt/telnet', async (req, res) => {
    const { host, port, username, password, command, tipe } = req.body;
    const connection = new Telnet();
    const params = { host, port: port || 23, shellPrompt: '>', timeout: 5000, username, password, loginPrompt: 'login: ', passwordPrompt: 'password: ' };
    try {
        await connection.connect(params);
        try { await connection.exec('enable'); } catch(e) {}
        
        const groupedOnu = {};
        const maxPorts = tipe === 'HIOSO' ? 2 : 4; // HIOSO 2 port, HSGQ 4 port

        for (let p = 1; p <= maxPorts; p++) {
            const portName = `PON ${p}`;
            groupedOnu[portName] = [];
            
            let cmdToRun = command;
            if (!command || command === 'show onu register-info') {
                cmdToRun = tipe === 'HIOSO' ? `show onu register epon 0/${p}` : `show onu register-info epon 0/${p}`;
            } else {
                cmdToRun = `${command} epon 0/${p}`;
            }

            try {
                let resCmd = await connection.exec(cmdToRun);
                const lines = resCmd.split('\n');
                lines.forEach(line => {
                    if (line.includes('MAC') || line.match(/([0-9a-fA-F]{2}:){5}[0-9a-fA-F]{2}/)) {
                        const macMatch = line.match(/([0-9a-fA-F]{2}:){5}[0-9a-fA-F]{2}/);
                        const status = line.toLowerCase().includes('online') || line.toLowerCase().includes('up') ? 'Online' : 'Offline';
                        const redamanMatch = line.match(/-\d+(\.\d+)?/);
                        groupedOnu[portName].push({
                            onu_index: groupedOnu[portName].length + 1,
                            mac_address: macMatch ? macMatch[0] : 'unknown',
                            status: status,
                            redaman: redamanMatch ? redamanMatch[0] + ' dBm' : 'Offline',
                            status_terakhir: new Date().toISOString().replace('T', ' ').substring(0, 19)
                        });
                    }
                });
            } catch (errPort) {
                console.error('Error port ' + p + ':', errPort.message);
            }
        }
        await connection.destroy();
        res.json({ success: true, parsed: groupedOnu });
    } catch (error) { res.status(500).json({ success: false, error: error.message }); }
});
app.listen(3000, () => console.log('OLT Proxy server running on port 3000'));
  1. Jalankan npm install express telnet-client cors lalu node olt-proxy.js.
  2. Masukkan alamat proxy http://localhost:3000/api/olt/telnet ke dalam input API Proxy Gateway di atas.
